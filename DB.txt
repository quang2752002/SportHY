/* ============================================================
   1. KHOI
   Ví dụ:
   - Giáo dục
   - Y tế
   - Văn hóa
   - Công an
   - Quân đội
   ============================================================ */

IF OBJECT_ID('dbo.Khoi', 'U') IS NOT NULL
    DROP TABLE dbo.Khoi;
GO

CREATE TABLE Khoi
(
    Id              BIGINT IDENTITY(1,1) PRIMARY KEY,
    Ma              VARCHAR(50) NOT NULL,
    Ten             NVARCHAR(200) NOT NULL,
    MoTa            NVARCHAR(1000) NULL,

    TrangThai       BIT NOT NULL DEFAULT 1,

    NgayTao        DATETIME2 NOT NULL DEFAULT GETDATE(),
    NgayCapNhat    DATETIME2 NULL,

    CONSTRAINT UQ_Khoi_Ma UNIQUE (Ma)
);
GO


/* ============================================================
   2. DON VI
   Có thể phân cấp:
   
   Khối Giáo dục
      ├── Sở GD
      ├── Trường A
      └── Trường B

   Đơn vị có thể có đơn vị cha.
   ============================================================ */

IF OBJECT_ID('dbo.DonVi', 'U') IS NOT NULL
    DROP TABLE dbo.DonVi;
GO

CREATE TABLE DonVi
(
    Id              BIGINT IDENTITY(1,1) PRIMARY KEY,

    Ma              VARCHAR(50) NOT NULL,
    Ten             NVARCHAR(300) NOT NULL,

    KhoiId          BIGINT NULL,
    DonViChaId      BIGINT NULL,

    LoaiDonVi       VARCHAR(30) NULL,

    DiaChi          NVARCHAR(500) NULL,
    NguoiDaiDien    NVARCHAR(200) NULL,
    SoDienThoai     VARCHAR(30) NULL,
    Email           VARCHAR(200) NULL,

    MoTa            NVARCHAR(1000) NULL,

    TrangThai       BIT NOT NULL DEFAULT 1,

    NgayTao         DATETIME2 NOT NULL DEFAULT GETDATE(),
    NgayCapNhat     DATETIME2 NULL,

    CONSTRAINT UQ_DonVi_Ma UNIQUE (Ma),

    CONSTRAINT FK_DonVi_Khoi
        FOREIGN KEY (KhoiId)
        REFERENCES Khoi(Id),

    CONSTRAINT FK_DonVi_DonViCha
        FOREIGN KEY (DonViChaId)
        REFERENCES DonVi(Id),

    CONSTRAINT CK_DonVi_Loai
        CHECK
        (
            LoaiDonVi IS NULL
            OR LoaiDonVi IN
            (
                'So',
                'Tinh',
                'Huyen',
                'Xa',
                'Truong',
                'BenhVien',
                'DoanhNghiep',
                'DonViTrucThuoc',
                'Khac'
            )
        )
);
GO


/* ============================================================
   3. GIAI DAU
   PhamVi:
   - TatCa
   - TheoKhoi

   Nếu TheoKhoi thì sử dụng bảng GiaiDauKhoi.
   ============================================================ */

IF OBJECT_ID('dbo.GiaiDau', 'U') IS NOT NULL
    DROP TABLE dbo.GiaiDau;
GO

CREATE TABLE GiaiDau
(
    Id              BIGINT IDENTITY(1,1) PRIMARY KEY,

    Ma              VARCHAR(50) NOT NULL,
    Ten             NVARCHAR(300) NOT NULL,

    MoTa            NVARCHAR(2000) NULL,

    NgayBatDau      DATE NOT NULL,
    NgayKetThuc     DATE NOT NULL,

    DiaDiem         NVARCHAR(500) NULL,

    PhamVi          VARCHAR(20) NOT NULL DEFAULT 'TatCa',

    TrangThai       VARCHAR(30) NOT NULL DEFAULT 'Nhap',

    NgayTao         DATETIME2 NOT NULL DEFAULT GETDATE(),
    NgayCapNhat     DATETIME2 NULL,

    CONSTRAINT UQ_GiaiDau_Ma UNIQUE (Ma),

    CONSTRAINT CK_GiaiDau_Ngay
        CHECK (NgayKetThuc >= NgayBatDau),

    CONSTRAINT CK_GiaiDau_PhamVi
        CHECK (PhamVi IN ('TatCa', 'TheoKhoi')),

    CONSTRAINT CK_GiaiDau_TrangThai
        CHECK
        (
            TrangThai IN
            (
                'Nhap',
                'SapDienRa',
                'DangDienRa',
                'KetThuc',
                'Huy'
            )
        )
);
GO

/* ============================================================
   8. NOI DUNG THI DAU
   Ví dụ:

   Điền kinh
      - 100m Nam
      - 100m Nữ
      - Nhảy xa Nam

   Cầu lông
      - Đơn nam
      - Đơn nữ
      - Đôi nam
      - Đôi nữ

   LoaiThiDau:
      CaNhan
      Doi
      DongDoi
   ============================================================ */

IF OBJECT_ID('dbo.NoiDungThiDau', 'U') IS NOT NULL
    DROP TABLE dbo.NoiDungThiDau;
GO

CREATE TABLE NoiDungThiDau
(
    Id                  BIGINT IDENTITY(1,1) PRIMARY KEY,

    GiaiDauMonTheThaoId BIGINT NOT NULL,

    Ma                  VARCHAR(50) NOT NULL,
    Ten                 NVARCHAR(200) NOT NULL,

    GioiTinh            VARCHAR(20) NOT NULL DEFAULT 'HonHop',

    LoaiThiDau          VARCHAR(30) NOT NULL DEFAULT 'CaNhan',

    SoLuongToiThieu     INT NULL,
    SoLuongToiDa        INT NULL,

    MoTa                NVARCHAR(1000) NULL,

    TrangThai           BIT NOT NULL DEFAULT 1,

    NgayTao             DATETIME2 NOT NULL DEFAULT GETDATE(),
    NgayCapNhat         DATETIME2 NULL,

    CONSTRAINT FK_NoiDungThiDau_GiaiDauMon
        FOREIGN KEY (GiaiDauMonTheThaoId)
        REFERENCES GiaiDauMonTheThao(Id),

    CONSTRAINT UQ_NoiDungThiDau
        UNIQUE (GiaiDauMonTheThaoId, Ma),

    CONSTRAINT CK_NoiDung_GioiTinh
        CHECK
        (
            GioiTinh IN
            (
                'Nam',
                'Nu',
                'HonHop'
            )
        ),

    CONSTRAINT CK_NoiDung_LoaiThiDau
        CHECK
        (
            LoaiThiDau IN
            (
                'CaNhan',
                'Doi',
                'DongDoi'
            )
        ),

    CONSTRAINT CK_NoiDung_SoLuong
        CHECK
        (
            SoLuongToiThieu IS NULL
            OR SoLuongToiDa IS NULL
            OR SoLuongToiDa >= SoLuongToiThieu
        )
);
GO

/* ============================================================
   20. TRAN DAU
   Không lưu Doi1 / Doi2.

   Vì:
   - Bóng đá: 2 đội
   - Điền kinh: 8 VĐV
   - Bơi: nhiều VĐV
   - Nhảy xa: nhiều VĐV
   - Các môn khác có thể có N người.

   Thành phần trận đấu nằm ở ThanhPhanTranDau.
   ============================================================ */

IF OBJECT_ID('dbo.TranDau', 'U') IS NOT NULL
    DROP TABLE dbo.TranDau;
GO

CREATE TABLE TranDau
(
    Id                  BIGINT IDENTITY(1,1) PRIMARY KEY,

    NoiDungThiDauId    BIGINT NOT NULL,
    VongDauId          BIGINT NOT NULL,

    BangDauId           BIGINT NULL,

    SanDauId            BIGINT NULL,

    SoTran              INT NOT NULL,

    TenTran             NVARCHAR(300) NULL,

    ThoiGianDuKien      DATETIME2 NULL,

    ThoiGianBatDau      DATETIME2 NULL,
    ThoiGianKetThuc     DATETIME2 NULL,

    TrangThai           VARCHAR(30) NOT NULL DEFAULT 'ChuaDau',

    GhiChu              NVARCHAR(1000) NULL,

    NgayTao             DATETIME2 NOT NULL DEFAULT GETDATE(),
    NgayCapNhat         DATETIME2 NULL,

    CONSTRAINT FK_TranDau_NoiDung
        FOREIGN KEY (NoiDungThiDauId)
        REFERENCES NoiDungThiDau(Id),

    CONSTRAINT FK_TranDau_Vong
        FOREIGN KEY (VongDauId)

Tôi đã cập nhật thuật toán xếp lịch tự động cho thể thức "Kết hợp vòng bảng & loại trực tiếp" (KetHopVongBangVaLoaiTrucTiep) trong 

TranDauService.cs
 và đồng bộ giao diện 

lich-thi-dau/page.tsx
.

Chi tiết các cải tiến đã thực hiện:
1. Tự động sinh nhánh đấu loại trực tiếp nối tiếp Vòng bảng
Sau khi sinh lịch thi đấu vòng tròn cho các bảng đấu (Round-Robin), hệ thống sẽ tự động tạo tiếp các vòng knockout tương ứng theo quy mô số bảng:

Trường hợp từ 4 bảng trở lên (Bảng A, B, C, D...):

Vòng Tứ kết (4 trận):
Tứ kết 1: Nhất Bảng A vs Nhì Bảng B
Tứ kết 2: Nhất Bảng C vs Nhì Bảng D
Tứ kết 3: Nhất Bảng B vs Nhì Bảng A
Tứ kết 4: Nhất Bảng D vs Nhì Bảng C
Vòng Bán kết (2 trận):
Bán kết 1: Thắng Tứ kết 1 vs Thắng Tứ kết 2
Bán kết 2: Thắng Tứ kết 3 vs Thắng Tứ kết 4
Trận Tranh hạng 3 - 4 (1 trận):
Thua Bán kết 1 vs Thua Bán kết 2
Trận Chung kết (1 trận):
Thắng Bán kết 1 vs Thắng Bán kết 2
Trường hợp 2 hoặc 3 bảng:

Vòng Bán kết (2 trận):
Với 2 bảng: Nhất Bảng A vs Nhì Bảng B, Nhất Bảng B vs Nhì Bảng A.
Với 3 bảng: Nhất Bảng A vs Nhì bảng có thành tích tốt nhất, Nhất Bảng B vs Nhất Bảng C.
Trận Tranh hạng 3 - 4 (1 trận): Thua Bán kết 1 vs Thua Bán kết 2.
Trận Chung kết (1 trận): Thắng Bán kết 1 vs Thắng Bán kết 2.
Trường hợp 1 bảng:

Nếu có từ 4 đội trở lên: Bán kết (Nhất vs Hạng 4, Nhì vs Hạng 3) $\rightarrow$ Tranh hạng 3 - 4 $\rightarrow$ Chung kết.
Nếu dưới 4 đội: Trận Chung kết (Nhất vs Nhì).
2. Ràng buộc thời gian & phân bổ sân bãi (CSP Engine)
Thứ tự vòng đấu (Round Precedence): Các trận Tứ kết bắt buộc bắt đầu sau khi Vòng bảng đã hoàn tất; Bán kết bắt đầu sau khi Tứ kết hoàn tất; Tranh 3-4 và Chung kết diễn ra sau Bán kết.
Ngày thi đấu: Nếu bật Ưu tiên Chung kết ngày bế mạc hoặc Mỗi vòng một ngày, các vòng đấu sẽ trải đều theo ngày giải đấu, trận Tranh hạng 3-4 và trận Chung kết sẽ diễn ra vào ngày bế mạc (trận Tranh 3-4 diễn ra trước trận Chung kết).
Phân bổ Sân đấu & Trọng tài: Các trận vòng knockout vẫn được xếp sân và phân công trọng tài tự động, đảm bảo không trùng sân và cân bằng tải trọng tài.
3. Hiển thị thông minh trên giao diện
Huy hiệu Nhánh đấu: Các trận knockout chưa có kết quả thực tế sẽ hiển thị định danh rõ ràng như Nhất Bảng A [Nhánh đấu] vs Nhì Bảng B [Nhánh đấu], hoặc Thắng Bán kết 1 [Nhánh đấu] vs Thắng Bán kết 2 [Nhánh đấu].
Bộ lọc vòng đấu: Người dùng có thể lọc theo từng vòng cụ thể: Vòng bảng (Lượt 1, 2...), Tứ kết, Bán kết, Tranh hạng 3 - 4, Chung kết.
Khi kết thúc vòng bảng, ban tổ chức có thể nhấn Sửa trận đấu để cập nhật đội tuyển thực tế đã giành quyền đi tiếp vào trận đấu.





Trong hệ thống xếp lịch tự động thi đấu, khi bạn chọn thể thức Vòng bảng + Loại trực tiếp, các vị trí này xuất hiện do cơ chế tính toán số đội đi tiếp để ghép vào cây thi đấu Knock-out:

1. Ý nghĩa của từng vị trí
Đội thứ 3 tốt nhất (1) và (2):

Đây là 2 đội xếp hạng 3 ở các bảng đấu có thành tích cao nhất (so sánh lần lượt theo: Điểm số $\rightarrow$ Hiệu số bàn thắng/thua $\rightarrow$ Số bàn thắng ghi được giữa các đội đứng thứ 3 với nhau).
Vé vớt 1 và Vé vớt 2:

Đây là 2 suất vé vớt bổ sung (Wildcard / Play-in). Nó xuất hiện khi giải đấu của bạn chỉ có 2 bảng đấu, nhưng vòng Knock-out lại được cấu hình là vòng Tứ kết (cần đúng 8 đội).
2. Tại sao lại xuất hiện cả hai cùng lúc?
Để tổ chức vòng Tứ kết, bắt buộc phải có đủ 8 đội (để chia thành 4 trận đấu).

Tuy nhiên, nếu giải của bạn chỉ có 2 bảng đấu (Bảng A & Bảng B):

2 đội Nhất (Nhất A, Nhất B) + 2 đội Nhì (Nhì A, Nhì B) = 4 đội.
Lấy thêm 2 đội hạng 3 (Hạng 3 bảng A, Hạng 3 bảng B) = 6 đội.
Vẫn thiếu 2 đội nữa mới đủ 8 đội cho 4 cặp Tứ kết!
Do đó, thuật toán tự động sinh thêm 2 suất giữ chỗ mang tên "Vé vớt 1" và "Vé vớt 2" (thực tế sẽ là 2 đội hạng 4 có thành tích tốt nhất, hoặc 2 đội thắng trận play-off tranh vé vớt) để ghép đủ 4 cặp đấu Tứ kết:

Tứ kết 1: Nhất A vs Nhì B
Tứ kết 2: Nhất B vs Đội thứ 3 tốt nhất (1)
Tứ kết 3: Vé vớt 1 vs Đội thứ 3 tốt nhất (2)
Tứ kết 4: Vé vớt 2 vs Nhì A
3. Gợi ý thiết lập hợp lý nhất cho giải đấu
Tùy vào quy mô giải mà bạn có thể áp dụng:

Nếu giải chỉ có 2 bảng:

Chuẩn và đẹp nhất là không lấy vé vớt/đội thứ 3, chỉ lấy 4 đội (Nhất A, Nhì A, Nhất B, Nhì B) vào thẳng Bán kết:
Bán kết 1: Nhất A vs Nhì B
Bán kết 2: Nhất B vs Nhì A
Hai đội thắng vào Chung kết, hai đội thua đá Tranh 3 - 4.
Khi đó sơ đồ sẽ vừa khít, không bao giờ xuất hiện chữ "Vé vớt".
Nếu muốn đá Tứ kết (8 đội):

Bạn nên chia giải thành 3 hoặc 4 bảng:
4 bảng (A, B, C, D): Lấy Nhất & Nhì mỗi bảng = đúng 8 đội vào Tứ kết.
3 bảng (A, B, C): Lấy 3 Nhất + 3 Nhì = 6 đội, cộng thêm đúng 2 đội thứ 3 tốt nhất (1 & 2) là tròn 8 đội Tứ kết (không bị dư ra Vé vớt 1, 2).
